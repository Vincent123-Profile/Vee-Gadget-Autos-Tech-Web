rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function isAdmin() {
      return request.auth != null && request.auth.token.admin == true;
    }

    // Public catalog media. Only verified administrators upload or replace files.
    match /products/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 10 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*|video/.*');
    }

    match /automobiles/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 25 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*|video/.*');
    }

    match /veexchange/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 15 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*|video/.*');
    }

    match /videos/{allPaths=**} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 100 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*|video/.*');
    }

    // Receipt uploads are not enabled in the current checkout flow.
    // Keep private until an authenticated, order-scoped receipt workflow is added.
    match /receipts/{allPaths=**} {
      allow read, write: if isAdmin();
    }
  }
}
